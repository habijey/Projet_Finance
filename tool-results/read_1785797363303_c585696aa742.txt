   131→function getDB() {
   132→  if (!dbPromise) {
   133→    dbPromise = openDB<MesDepensesDB>("mesdepenses-db", 2, {
   134→      upgrade(db, oldVersion) {
   135→        if (!db.objectStoreNames.contains("settings")) {
   136→          db.createObjectStore("settings", { keyPath: "id" })
   137→        }
   138→        if (!db.objectStoreNames.contains("categories")) {
   139→          const catStore = db.createObjectStore("categories", { keyPath: "id" })
   140→          catStore.createIndex("by-name", "name", { unique: true })
   141→        }
   142→        if (!db.objectStoreNames.contains("expenses")) {
   143→          const expStore = db.createObjectStore("expenses", { keyPath: "id" })
   144→          expStore.createIndex("by-date", "date")
   145→          expStore.createIndex("by-category", "categoryId")
   146→        }
   147→        if (!db.objectStoreNames.contains("savingsGoals")) {
   148→          db.createObjectStore("savingsGoals", { keyPath: "id" })
   149→        }
   150→        if (!db.objectStoreNames.contains("incomes")) {
   151→          const incStore = db.createObjectStore("incomes", { keyPath: "id" })
   152→          incStore.createIndex("by-date", "date")
   153→        }
   154→        if (!db.objectStoreNames.contains("subscriptions")) {
   155→          const subStore = db.createObjectStore("subscriptions", { keyPath: "id" })
   156→          subStore.createIndex("by-active", "active")
   157→        }
   158→      },
   159→    })
   160→  }
   161→  return dbPromise
   162→}

   265→export async function getExpenses(opts?: {
   266→  month?: string
   267→  categoryId?: string
   268→  search?: string
   269→  minAmount?: number
   270→  maxAmount?: number
   271→  sortBy?: "date" | "amount"
   272→  sortOrder?: "asc" | "desc"
   273→}): Promise<Expense[]> {
   274→  const db = await getDB()
   275→  let expenses = await db.getAll("expenses")
   276→
   277→  // Filter by month
   278→  if (opts?.month) {
   279→    const [year, month] = opts.month.split("-").map(Number)
   280→    const startDate = new Date(year, month - 1, 1).getTime()
   281→    const endDate = new Date(year, month, 0, 23, 59, 59, 999).getTime()
   282→    expenses = expenses.filter((e) => {
   283→      const t = new Date(e.date).getTime()
   284→      return t >= startDate && t <= endDate
   285→    })
   286→  }
   287→
   288→  // Filter by category
   289→  if (opts?.categoryId) {
   290→    expenses = expenses.filter((e) => e.categoryId === opts.categoryId)
   291→  }
   292→
   293→  // Search
   294→  if (opts?.search) {
   295→    const q = opts.search.toLowerCase()
   296→    expenses = expenses.filter(
   297→      (e) => e.description.toLowerCase().includes(q) || (e.note?.toLowerCase().includes(q))
   298→    )
   299→  }
   300→
   301→  // Amount range
   302→  if (opts?.minAmount !== undefined) {
   303→    expenses = expenses.filter((e) => e.amount >= opts.minAmount!)
   304→  }
   305→  if (opts?.maxAmount !== undefined) {
   306→    expenses = expenses.filter((e) => e.amount <= opts.maxAmount!)
   307→  }
   308→
   309→  // Sort
   310→  const sortBy = opts?.sortBy || "date"
   311→  const sortOrder = opts?.sortOrder || "desc"
   312→  expenses.sort((a, b) => {
   313→    const aVal = sortBy === "date" ? new Date(a.date).getTime() : a.amount
   314→    const bVal = sortBy === "date" ? new Date(b.date).getTime() : b.amount
   315→    return sortOrder === "asc" ? aVal - bVal : bVal - aVal
   316→  })
   317→
   318→  // Attach category objects
   319→  const categories = await db.getAll("categories")
   320→  const catMap = new Map(categories.map((c) => [c.id, c]))
   321→  for (const e of expenses) {
   322→    e.category = e.categoryId ? catMap.get(e.categoryId) || null : null
   323→  }
   324→
   325→  return expenses
   326→}

   523→// Generate pending subscription expenses (call on app init)
   524→export async function generateSubscriptionExpenses(): Promise<Expense[]> {
   525→  const db = await getDB()
   526→  const subscriptions = await db.getAll("subscriptions")
   527→  const allExpenses = await db.getAll("expenses")
   528→  const today = new Date().toISOString().slice(0, 10)
   529→  // Track which (subscriptionId:occurrence) combos already exist
   530→  const done = new Set(
   531→    allExpenses
   532→      .filter((e: any) => e.subscriptionId && e.occurrence)
   533→      .map((e: any) => `${e.subscriptionId}:${e.occurrence}`)
   534→  )
   535→  const created: Expense[] = []
   536→  const tx = db.transaction("expenses", "readwrite")
   537→  for (const sub of subscriptions) {
   538→    if (!sub.active) continue
   539→    const startIso = (sub.createdAt || now).slice(0, 10)
   540→    const occurrences = dueOccurrences(sub, startIso, today)
   541→    for (const occ of occurrences) {
   542→      if (done.has(`${sub.id}:${occ}`)) continue
   543→      const now = new Date().toISOString()
   544→      const expense: Expense = {
   545→        id: uid(),
   546→        amount: sub.amount,
   547→        description: sub.name,
   548→        date: new Date(occ).toISOString(),
   549→        categoryId: sub.categoryId || undefined,
   550→        category: null,
   551→        note: "Prélèvement automatique",
   552→        createdAt: now,
   553→        updatedAt: now,
   554→        subscriptionId: sub.id,
   555→        occurrence: occ,
   556→      }
   557→      await tx.store.put(expense)
   558→      created.push(expense)
   559→    }
   560→  }
   561→  await tx.done()
   562→  return created
   563→}
   564→
   565→function dueOccurrences(sub: Subscription, fromIso: string, toIso: string): string[] {
   566→  if (fromIso > toIso) return []
   567→  const [fy, fm] = fromIso.split("-").map(Number)
   568→  const [ty, tm] = toIso.split("-").map(Number)
   569→  const yearly = sub.frequency === "yearly"
   570→  const out: string[] = []
   571→  for (let index = fy * 12 + (fm - 1); index <= ty * 12 + (tm - 1); index++) {
   572→    const year = Math.floor(index / 12)
   573→    const month = (index % 12) + 1
   574→    const daysInMonth = new Date(year, month, 0).getDate()
   575→    const day = Math.min(sub.dayOfMonth, daysInMonth)
   576→    const date = `${year}-${String(month).padStart(2, "0")}-${String(day).padStart(2, "0")}`
   577→    if (date < fromIso || date > toIso) continue
   578→    if (date > today) continue
   579→    out.push(date)
   580→  }
   581→  return out
   582→}

   586→export async function importCSV(text: string): Promise<{ imported: number; errors: string[]; total: number }> {
   587→  const db = await getDB()
   588→  const categories = await db.getAll("categories")
   589→  const catMap = new Map(categories.map((c) => [c.name.toLowerCase(), c]))
   590→
   591→  const lines = text.split("\n").filter((l) => l.trim())
   592→  let imported = 0
   593→  const errors: string[] = []
   594→
   595→  const tx = db.transaction("expenses", "readwrite")
   596→
   597→  for (let i = 1; i < lines.length; i++) {
   598→    const line = lines[i].trim()
   599→    if (!line) continue
   600→
   601→    // Try semicolon then comma
   602→    let parts = line.split(";").map((p) => p.trim().replace(/^"|"$/g, ""))
   603→    if (parts.length < 3) {
   604→      parts = line.split(",").map((p) => p.trim().replace(/^"|"$/g, ""))
   605→    }
   606→    if (parts.length < 3) {
   607→      errors.push(`Ligne ${i + 1}: format invalide`)
   608→      continue
   609→    }
   610→
   611→    const dateStr = parts[0]
   612→    const description = parts[1]
   613→    const amountStr = parts[2].replace(",", ".").replace(/[^\d.\-]/g, "")
   614→    const categoryName = parts[3] || ""
   615→
   616→    const amount = parseFloat(amountStr)
   617→    if (isNaN(amount)) {
   618→      errors.push(`Ligne ${i + 1}: montant invalide "${parts[2]}"`)
   619→      continue
   620→    }
   621→
   622→    let date: Date
   623→    const dmy = dateStr.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/)
   624→    if (dmy) {
   625→      date = new Date(parseInt(dmy[3]), parseInt(dmy[2]) - 1, parseInt(dmy[1]))
   626→    } else {
   627→      const ymd = dateStr.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/)
   628→      if (ymd) {
   629→        date = new Date(parseInt(ymd[1]), parseInt(ymd[2]) - 1, parseInt(ymd[3]))
   630→      } else {
   631→        date = new Date()
   632→      }
   633→    }
   634→
   635→    const cat = categoryName ? catMap.get(categoryName.toLowerCase()) : undefined
   636→    const now = new Date().toISOString()
   637→
   638→    const expense: Expense = {
   639→      id: uid(),
   640→      amount: Math.abs(amount),
   641→      description,
   642→      date: date.toISOString(),
   643→      categoryId: cat?.id,
   644→      category: cat || null,
   645→      note: null,
   646→      createdAt: now,
   647→      updatedAt: now,
   648→    }
   649→    await tx.store.put(expense)
   650→    imported++
   651→  }
   652→
   653→  await tx.done
   654→  return { imported, errors, total: lines.length - 1 }
   655→}

   718→export async function importFullBackup(jsonStr: string): Promise<{
   719→  settings: Settings
   720→  categories: Category[]
   721→  expenses: Expense[]
   722→  savingsGoals: SavingsGoal[]
   723→  incomes: Income[]
   724→  subscriptions: Subscription[]
   725→}> {
   726→  const backup: BackupData = JSON.parse(jsonStr)
   727→
   728→  if (!backup.version || !backup.settings || !backup.categories) {
   729→    throw new Error("Fichier de sauvegarde invalide")
   730→  }
   731→
   732→  const db = await getDB()
   733→
   734→  // Clear everything
   735→  await db.clear("expenses")
   736→  await db.clear("savingsGoals")
   737→  await db.clear("categories")
   738→  await db.clear("settings")
   739→  await db.clear("subscriptions")
   740→
   741→  // Restore settings
   742→  await db.put("settings", backup.settings)
   743→
   744→  // Restore categories (strip category from expenses to avoid FK issues)
   745→  const expenses: Expense[] = (backup.expenses || []).map((e) => ({
   746→    ...e,
   747→    category: undefined,
   748→  }))
   749→  const categories: Category[] = backup.categories || []
   750→
   751→  // Restore savings goals
   752→  const savingsGoals: SavingsGoal[] = backup.savingsGoals || []
   753→  const incomes: Income[] = backup.incomes || []
   754→  const subscriptions: Subscription[] = backup.subscriptions || []
   755→
   756→  // Write everything in a transaction
   757→  const tx = db.transaction(["settings", "categories", "expenses", "savingsGoals", "incomes", "subscriptions"], "readwrite")
   758→
   759→  await tx.objectStore("settings").put(backup.settings)
   760→  for (const cat of categories) {
   761→    await tx.objectStore("categories").put(cat)
   762→  }
   763→  for (const exp of expenses) {
   764→    await tx.objectStore("expenses").put(exp)
   765→  }
   766→  for (const goal of savingsGoals) {
   767→    await tx.objectStore("savingsGoals").put(goal)
   768→  }
   769→  for (const inc of incomes) {
   770→    await tx.objectStore("incomes").put(inc)
   771→  }
   772→  for (const sub of subscriptions) {
   773→    await tx.objectStore("subscriptions").put(sub)
   774→  }
   775→
   776→  await tx.done
   777→
   778→  return { settings: backup.settings, categories, expenses, savingsGoals, incomes, subscriptions }
   779→}

   783→export type AdviceSeverity = "critical" | "warning" | "info" | "good"
   784→
   785→export interface Advice {
   786→  id: string
   787→  severity: AdviceSeverity
   788→  text: string
   789→}
   790→
   791→export async function computeAdvice(
   792→  categories: Category[],
   793→  expenses: Expense[],
   794→  incomes: Income[],
   795→  settings: Settings
   796→): Promise<Advice[]> {
   797→  const advice: Advice[] = []
   798→  const totalExpenses = expenses.reduce((s, e) => s + e.amount, 0)
   799→  const totalIncomes = incomes.reduce((s, i) => s + i.amount, 0) + (settings.monthlySalary || 0)
   800→
   801→  // 1. Budget alerts
   802→  for (const cat of categories) {
   803→    if (cat.budgetLimit <= 0) continue
   804→    const spent = expenses.filter((e) => e.categoryId === cat.id).reduce((s, e) => s + e.amount, 0)
   805→    const pct = (spent / cat.budgetLimit) * 100
   806→    if (pct >= 100) {
   807→      advice.push({
   808→        id: `over-${cat.id}`,
   809→        severity: "critical",
   810→        text: `Budget « ${cat.name} » dépassé : ${formatEUR(spent)} sur ${formatEUR(cat.budgetLimit)}.`,
   811→      })
   812→    } else if (pct >= 80) {
   813→      advice.push({
   814→        id: `warn-${cat.id}`,
   815→        severity: "warning",
   816→        text: `Budget « ${cat.name} » consommé à ${Math.round(pct)} %. Garde un œil.`,
   817→      })
   818→    }
   819→  }
   820→
   821→  // 2. Negative savings
   822→  if (totalIncomes > 0 && totalExpenses > totalIncomes) {
   823→    const diff = totalExpenses - totalIncomes
   824→    advice.push({
   825→      id: "negative-savings",
   826→      severity: "critical",
   827→      text: `Ce mois, tu dépenses plus que tes revenus (${formatEUR(diff)} de plus).`,
   828→    })
   829→  }
   830→
   831→  // 3. Good savings
   832→  if (totalIncomes > 0) {
   833→    const rate = Math.round(((totalIncomes - totalExpenses) / totalIncomes) * 100)
   834→    if (rate >= 20) {
   835→      advice.push({
   836→        id: "good-savings",
   837→        severity: "good",
   838→        text: `Beau taux d'épargne : ${rate} % de tes revenus mis de côté ce mois-ci.`,
   839→      })
   840→    }
   841→  }
   842→
   843→  // 4. Suggest budget for top unbudgeted category
   844→  const budgeted = new Set(categories.filter((c) => c.budgetLimit > 0).map((c) => c.id))
   845→  const topUnbudgeted = categories
   846→    .filter((c) => !budgeted.has(c.id))
   847→    .map((c) => ({
   848→      cat: c,
   849→      total: expenses.filter((e) => e.categoryId === c.id).reduce((s, e) => s + e.amount, 0),
   850→    }))
   851→    .filter((x) => x.total >= 50)
   852→    .sort((a, b) => b.total - a.total)[0]
   853→  if (topUnbudgeted) {
   854→    advice.push({
   855→      id: `suggest-${topUnbudgeted.cat.id}`,
   856→      severity: "info",
   857→      text: `« ${topUnbudgeted.cat.name} » représente ${formatEUR(topUnbudgeted.total)} ce mois sans budget défini.`,
   858→    })
   859→  }
   860→
   861→  // 5. Large expense alert
   862→  if (settings.largeExpenseAlert > 0) {
   863→    const largeExpenses = expenses.filter((e) => e.amount >= settings.largeExpenseAlert)
   864→    if (largeExpenses.length > 0) {
   865→      advice.push({
   866→        id: "large-expenses",
   867→        severity: "info",
   868→        text: `${largeExpenses.length} dépense${largeExpenses.length > 1 ? "s" : ""} supérieure${largeExpenses.length > 1 ? "s" : ""} à ${formatEUR(settings.largeExpenseAlert)} ce mois.`,
   869→      })
   870→    }
   871→  }
   872→
   873→  const order: Record<AdviceSeverity, number> = { critical: 0, warning: 1, info: 2, good: 3 }
   874→  return advice.sort((a, b) => order[a.severity] - order[b.severity])
   875→}
   876→
   877→function formatEUR(n: number): string {
   878→  return new Intl.NumberFormat("fr-FR", { style: "currency", currency: "EUR" }).format(n)
   879→}
