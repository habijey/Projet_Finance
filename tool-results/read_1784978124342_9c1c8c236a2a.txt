   129→function formatMoney(n: number): string {
   130→  return new Intl.NumberFormat("fr-FR", {
   131→    style: "currency",
   132→    currency: "EUR",
   133→  }).format(n)
   134→}
   135→
   136→function formatDate(d: string): string {
   137→  return new Intl.DateTimeFormat("fr-FR", {
   138→    day: "numeric",
   139→    month: "long",
   140→    year: "numeric",
   141→  }).format(new Date(d))
   142→}
   143→
   144→function formatDateShort(d: string): string {
   145→  return new Intl.DateTimeFormat("fr-FR", {
   146→    day: "numeric",
   147→    month: "short",
   148→  }).format(new Date(d))
   149→}
   150→
   151→function getCurrentMonthStr(): string {
